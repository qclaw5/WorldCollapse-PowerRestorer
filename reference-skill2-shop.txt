function updateSkill2(dt){
  const p=G.player; if(!p||!p.s2) return;
  const s=p.s2;
  if(s.cd>0) s.cd=Math.max(0,s.cd-dt);
  if(p.s2haste>0) p.s2haste=Math.max(0,p.s2haste-dt);
  if(p.chainSlam){
    p.chainSlam.t+=dt;
    if(p.chainSlam.t>=p.chainSlam.dur) p.chainSlam=null;
  }
  if(s.phase!=='active') return;
  const def=s2Def(p); if(!def){ s2Clear(p,false); return; }
  s.t+=dt;
  if(def.tick) def.tick(p,s,dt);
  if(s.phase!=='active') return;
  if(s.t>=s.dur){ if(def.expire) def.expire(p,s); s2Clear(p,false); }
}

function shopState(){
  const d=Save.data;
  if(!d.shop || typeof d.shop!=='object' || Array.isArray(d.shop)) d.shop=freshShopState();
  const s=d.shop;
  if(!Number.isFinite(s.seq)) s.seq=0;
  if(!Number.isFinite(s.nextAt)) s.nextAt=0;
  if(!Array.isArray(s.listing)) s.listing=[];
  if(!s.stock || typeof s.stock!=='object' || Array.isArray(s.stock)) s.stock={};
  /* 只保留合法 SKU，防止旧存档 / 手改数据混入 */
  const valid=id=>!!(GEAR_MAP[id]||CONSUM_MAP[id]);
  s.listing=[...new Set(s.listing.filter(valid))].slice(0,MERCHANT.slots);
  const st={};
  for(const id of s.listing){
    const v=s.stock[id];
    st[id]=(v===-1||v===null||v===Infinity)?-1:clamp(Math.floor(Number(v)||0),0,99);
  }
  s.stock=st;
  return s;
}

function gearConsumableSummary(){
    let h='<div class="gear-sect">神 秘 商 人<small>兜帽下只看得见一枚发亮的硬币。每轮只上 '+MERCHANT.slots+' 种货，买完即售罄，定时补货或用宝石刷新。</small></div>';

    h+='<div class="gear-note">金币由关卡小怪<b>必定掉落</b>，怪物等级越高掉得越多；等级 ≥5 的怪物有概率掉落宝石，Boss 必掉。换图 / 结算时场上未拾取的货币会自动入账。</div>';
    h+='<div class="gear-note"><b>消耗品</b>：药水在战斗中按 B 饮用，卷轴使用后<b>记录招式并自动销毁</b>。同类药水刷新不叠加，最多同时 4 种增益；卷轴已记录后不再上架。</div>';
    const inv=Save.data.items||{};
    for(const cur of ['coin','gem']){
      const gears=GEAR_DEFS.filter(x=>x.cur===cur&&s.listing.includes(x.id));
      const cons=CONSUMABLES.filter(c=>c.cur===cur&&s.listing.includes(c.id));
      h+='<div class="gear-sect" style="margin-top:9px">'+(cur==='coin'?'金 币 货 架':'宝 石 货 架 · 高 级 道 具')+
         '<small>本轮上架 '+(gears.length+cons.length)+' 种</small></div><div class="gear-list">';
      if(!gears.length && !cons.length) h+='<div class="gitem"><div class="ds">本轮该货架无货 · 等待补货或用宝石刷新</div></div>';
      for(const g of gears){
        const owned=Save.data.owned.includes(g.id), eq=Save.data.equip[g.slot]===g.id;
        const afford=(Save.data[cur==='gem'?'gems':'coins']||0)>=g.cost;
        const chk=merchantCanBuy(g.id), sold=!owned&&!chk.ok&&chk.why==='sold';
        let btn;
        if(owned) btn=(eq?'<button disabled>已 装 备</button>':'<button class="pri" data-act="equip" data-id="'+g.id+'">装 备</button>');
        else if(sold) btn='<button disabled>售 罄 · 待 补 货</button>';
        else btn=(afford?'<button class="pri" data-act="buy" data-id="'+g.id+'">购 买</button>'
                        :'<button data-act="buy" data-id="'+g.id+'" disabled>货 币 不 足</button>');
        h+='<div class="gitem'+(eq?' eq':'')+(sold?' sold':'')+'"><div class="r1">'+
           '<div class="nm" style="color:'+g.color+'">'+g.name+' <span style="color:#9a94b8;font-size:.85em">· '+GEAR_SLOTS.find(z=>z.id===g.slot).name+'</span>'+
             (owned?'<span class="stock-tag inf">已 购 入</span>':merchantStockTag(g.id))+'</div>'+
           '<div class="pr'+(cur==='gem'?' gm':'')+'">'+g.cost+(cur==='gem'?' ◆':' G')+'</div></div>'+
           '<div class="ds">'+g.desc+'</div>'+armorBalanceLine(g)+'<div class="btns">'+btn+'</div></div>';
      }
  }
  // 未持有的消耗品也能在局内直接补货（方便连续测试）
  h+='<div class="bag-sect" style="margin-top:9px">现 场 补 货<small>用金币 / 宝石直接补，买完立刻能用</small></div>';
  for(const cur of ['coin','gem']){
    for(const d of CONSUMABLES.filter(c=>c.cur===cur && !(inv[c.id]>0))){
      h+=consumableLine(d,inv,inBattle);
    }
  }
  body.innerHTML=h;
}
function bagAction(e){
  const b=e.target.closest('button[data-act]'); if(!b) return;
  const id=b.dataset.id, act=b.dataset.act;
  if(act==='buy') buyConsumable(id);
  else if(act==='use'){
    const d=CONSUM_MAP[id]; if(!d) return;
    if(d.kind==='scroll') useScroll(id); else usePotion(id);
  }
  renderBagPanel(); updateBagBadge(); updateBuffHud();
}
