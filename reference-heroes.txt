/* ---------- [2] 二技能辅助函数（freshS2 / s2Find / s2PickTarget / s2HitArea / s2Burst / s2Ring / 各结局函数） · 原文 6488-6603 ---------- */
/* ============================================================
   二技能（可选）：每个权能一个专属技能，R / 按键区上方圆钮触发
   业报之权：发射锁链束缚抓取 → 再按一次拉回 → 自动衔接带强击退的普攻
   ============================================================ */
function freshS2(){ return { cd:0, phase:'idle', t:0, dur:0, data:null, fx:null }; }
function s2Def(p){ return SKILL2[p.pw.id]; }
function s2Find(id){
  if(id==null) return null;
  return G.enemies.find(e=>e.id===id && !e.dead && e.hp>0) || null;
}
function s2Clear(p,hard){
  const s=p.s2; if(!s) return;
  const d=s.data;
  if(d && d.target!=null){ const e=G.enemies.find(z=>z.id===d.target); if(e&&e.chain) e.chain=null; }
  s.phase='idle'; s.t=0; s.dur=0; s.data=null; s.fx=null;
  if(hard){ s.cd=0; p.chainSlam=null; p.s2haste=0; }
}
function s2Reset(p){ if(p&&p.s2) s2Clear(p,true); }
function s2PickTarget(p,maxD){
  maxD=maxD||250;
  const q=p.tgt && !p.tgt.dead && p.tgt.hp>0 ? p.tgt : null;
  if(q && dist2(q.x,q.y,p.x,p.y)<maxD*maxD) return q;
  let best=null,bd=maxD*maxD;
  for(const e of G.enemies){
    if(e.dead||e.hp<=0) continue;
    const d2=dist2(e.x,e.y,p.x,p.y);
    if(d2<bd){ bd=d2; best=e; }
  }
  return best;
}
function s2AimPoint(p,dist){
  const a=p.aimAng;
  return { x:clamp(p.x+Math.cos(a)*dist,16,G.arena.w-16),
           y:clamp(p.y+Math.sin(a)*dist,20,G.arena.h-16) };
}
function s2HitArea(x,y,r,dmg,opt){
  opt=opt||{}; let n=0;
  for(const e of G.enemies){
    if(e.dead||e.hp<=0) continue;
    const rr=r+(e.r||8);
    if(dist2(e.x,e.y-8,x,y)<rr*rr){ damageEnemy(e,dmg,opt); n++; }
  }
  return n;
}
function s2Burst(x,y,col,n,sp){
  for(let i=0;i<(n||10);i++)
    G.parts.push(mkPart(x+rand(-5,5),y+rand(-5,5),i%3===0?'#ffffff':col,
      {sp:rand(30,sp||120),life:rand(.2,.5),size:i%4===0?2:1,g:20}));
}
function s2Ring(x,y,r,col,dur){ G.slashes.push({x,y,a:0,arc:TAU,r,t:0,dur:dur||.4,color:col}); }
function s2KarmaEnd(p,s,expired){
  const d=s.data; if(!d) return;
  const e=s2Find(d.target);
  if(e){
    if(e.chain) e.chain=null;
    if(d.phase==='bound'){
      damageEnemy(e,expired?14:12,{noKb:true});
      s2Burst(e.x,e.y-8,expired?'#8c7aa5':'#c9a6ff',8,80);
      s2Ring(e.x,e.y-8,14,'#8c7aa5',.3);
    }
  }
  s2Clear(p,false);
}
function s2LifeBurst(p,s,manual){
  const d=s.data; if(!d) return;
  const e=s2Find(d.target);
  const x=e?e.x:(d.phase==='self'?p.x:d.x), y=e?e.y-8:(d.phase==='self'?p.y-8:d.y);
  if(e&&e.chain) e.chain=null;
  s2HitArea(x,y,manual?74:56,manual?46:26,{kbMul:manual?1.3:1});
  p.hp=Math.min(p.hpMax,p.hp+(manual?12:6));
  s2Burst(x,y,'#58e08a',manual?16:9,120);
  s2Ring(x,y,manual?30:20,'#c8ffdc',.45);
  flash('#58e08a',.18);
  s2Clear(p,false);
}
function s2FateOpen(p,s,x,y){
  const d=s.data, face=d.face;
  d.phase='done'; d.t=0; d.x=x; d.y=y;
  s2Burst(x,y,'#ff6fa8',18,160); s2Ring(x,y,26,'#ffd0e4',.45);
  flash('#ff6fa8',.22); G.shake=Math.max(G.shake,4);
  if(face>=4){
    s2HitArea(x,y,92,90,{big:true,kbMul:1.2});
    for(const e of G.enemies) if(!e.dead&&dist2(e.x,e.y-8,x,y)<92*92) e.burn=Math.max(e.burn,3);
    G.slashes.push({x,y,a:0,arc:TAU,r:52,t:0,dur:.5,color:'#ffe9a8'});
    toast('大点数 · 裁决爆发 '+face);
  } else {
    for(const e of G.enemies){
      if(e.dead||e.hp<=0) continue;
      damageEnemy(e,34,{});
      if(Math.random()<.5) e.slow=Math.max(e.slow,2); else e.burn=Math.max(e.burn,2);
    }
    addFatePoints(2);
    s2Ring(x,y,44,'#ff6fa8',.5);
    toast('小点数 · 命运累积 '+face);
  }
}
function s2OrderFall(p,s,manual){
  const d=s.data; if(!d) return;
  const x=d.x,y=d.y-8;
  s2HitArea(x,y,92,manual?78:40,{kbMul:manual?1.6:1,big:true});
  for(let i=0;i<10;i++)
    G.parts.push(mkPart(x+rand(-7,7),y-42+i*9,i%2?'#ffffff':'#ffe9a8',{sp:10,life:.4,size:2,g:0}));
  s2Burst(x,y,'#ffe9a8',manual?18:10,140);
  s2Ring(x,y,manual?34:22,'#ffe9a8',.45);
  flash('#cfe4ff',.18);
  s2Clear(p,false);
}
function s2VoidBurst(p,s,manual){
  const d=s.data; if(!d) return;
  const x=p.x,y=p.y-8,r=manual?110:84;
  s2HitArea(x,y,r,manual?64:32,{trueDmg:true,kbMul:manual?1.3:1,big:true});
  s2Burst(x,y,'#b07bff',manual?20:12,150);
  s2Ring(x,y,manual?44:30,'#e0d4ff',.5);
  flash('#b07bff',.22);
  s2Clear(p,false);
}

function updateSkill2(dt){
  const p=G.player; if(!p||!p.s2) return;
  const s=p.s2;
  if(s.cd>0) s.cd=Math.max(0,s.cd-dt);
  if(p.s2haste>0) p.s2haste=Math.max(0,p.s2haste-dt);
  if(p.chainSlam){
    p.chainSlam.t+=dt;
    if(p.chainSlam.t>=p.chainSlam.dur) p.chainSlam=null;
  }
  if(s.phase!=='active') return;
  const def=s2Def(p); if(!def){ s2Clear(p,false); return; }
  s.t+=dt;
  if(def.tick) def.tick(p,s,dt);
  if(s.phase!=='active') return;
  if(s.t>=s.dur){ if(def.expire) def.expire(p,s); s2Clear(p,false); }
}

/* 角色特效自检：几何字段有限、三档强度可用、全状态渲染不产生 NaN / 异常 */
function fxAudit(){
  const issues=[], C=CanvasRenderingContext2D.prototype;
  const names=['fillRect','arc','ellipse','drawImage','moveTo','lineTo','strokeRect','rect'];
  const orig={}, bad=[];
  for(const n of names){
    orig[n]=C[n];
    C[n]=function(...a){
      for(const v of a) if(typeof v==='number' && !Number.isFinite(v)){ bad.push(n); break; }
      return orig[n].apply(this,a);
    };
  }
  const geoKeys=['grip','r0','r1','oc','ow','tipOff'];
  for(const id of Object.keys(WEAPON_GEO)){
    const g=WEAPON_GEO[id];
    for(const k of geoKeys) if(g[k]!=null && !Number.isFinite(g[k])) issues.push(id+'.'+k+' 非有限值');
  }
  const savedTier=Settings.data.fxTier;
  const savedEnemies=G.enemies, savedScreen=G.screen, savedPower=G.power, savedUlt=G.ult;
  const q0=G.player, playerSnap=q0?Object.assign({},q0):null;
  let rendered=0, thrown=0;
  try{
    for(const tier of [0,1,2]){
      Settings.data.fxTier=tier;
      if(!Number.isFinite(fxScale())) issues.push('tier '+tier+' 强度系数异常');
    }
    Settings.data.fxTier=1;
    G.screen='battle';
    const types=Object.keys(ETYPES);
    G.enemies=[];
    for(const type of types){
      let e; try{ e=spawnEnemy(type); }catch(err){ issues.push(type+' 生成失败 '+err.message); continue; }
      e.spawnT=0; e.x=200; e.y=300; e.aimX=1; e.aimY=0;
      Object.assign(e,{burn:3,slow:3,weak:3,charging:.4,blinkWarn:.2,blinkIn:.15,level:9,hurt:.06});
    }
    for(const P of POWERS){
      G.power=P.id; G.ult=ULTS[0].id;
      if(!G.player) break;   /* 尚未进入战斗：只做静态校验，不算异常 */
      const q=G.player;
      Object.assign(q,{hurtFlash:.1,hitShake:5,shield:40,shieldFlash:1,skillT:1.5,
        karma:8,swing:.7,atkAng:.5,aimAng:.5,dashT:0,
        buffs:[{id:'p_edge',color:'#ff9db0',until:G.time+9,dur:12,fx:{dmg:1.2}},
               {id:'p_ward',color:'#ffd76b',until:G.time+9,dur:12,fx:{taken:.8}}],
        gearFx:{gale:true}});
      bad.length=0;
      try{ renderBattle(); rendered++; }catch(err){ thrown++; issues.push(P.id+' 渲染异常 '+err.message); }
      if(bad.length) issues.push(P.id+' 产生 NaN 参数 '+[...new Set(bad)].join(','));
    }
  } catch(err){ issues.push('自检异常 '+err.message); }
  finally{
    for(const n of names) C[n]=orig[n];
    Settings.data.fxTier=savedTier; G.enemies=savedEnemies; G.screen=savedScreen;
    G.power=savedPower; G.ult=savedUlt;
    if(q0 && playerSnap) for(const k of Object.keys(playerSnap)) q0[k]=playerSnap[k];
  }
  return { ok:issues.length===0, issues, powers:POWERS.length, enemyTypes:Object.keys(ETYPES).length,
    rendered, thrown, tier:['简洁','标准','华丽'][fxTier()], scale:+fxScale().toFixed(2) };
}
